  // 获取画面平均亮度
  function getBrightness(videoEl) {
  const w = videoEl.videoWidth;
  const h = videoEl.videoHeight;
  if(!w || !h || videoEl.readyState < 2) return lastBrightness;
  const canvas = document.createElement('canvas');
  canvas.width = w;
  canvas.height = h;
  const ctx = canvas.getContext('2d');
  ctx.drawImage(videoEl,0,0,w,h);
  const imgData = ctx.getImageData(0,0,w,h);
  let sum = 0;
  const data = imgData.data;
  for(let i=0;i<data.length;i+=4){
  let r = data[i];
  let g = data[i+1];
  let b = data[i+2];
  sum += (r*0.299 + g*0.587 + b*0.114);
  }
  return sum / (data.length/4);
  }

  analyser.getByteFrequencyData(dataArray);
  let sum = 0;
  for(let i=0;i<bufferLength;i++) sum += dataArray[i];
  let avg = sum / bufferLength;
  return avg;
  }
  // 自动循环检测
  function autoLoop(){
  if(!autoMode) return;
  drawId = requestAnimationFrame(autoLoop);
  let bright = getBrightness(video);
  let volume = getVolume();
  if(!measuring && !finished){
  if(bright > lastBrightness + lightThreshold){
  startTimestamp = Date.now();
  measuring = true;
  updateStatusMessage("⚡ 摄像头检测到闪光（闪电），等待雷声...");
  timeDiffSpan.innerText = "⏳ 识别雷声中";
  distanceSpan.innerText = "等待收音";
  }
  }
  if(measuring && !finished){
  if(volume > volThreshold){
  calculateAndDisplay(Date.now());
  }
  }
  lastBrightness = bright;
  }

  });
